cmake_minimum_required(VERSION 3.10)
project(msckf_vio)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_compile_options(-std=c++17)

# Modify cmake module path if new .cmake files are required
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_LIST_DIR}/cmake")

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(image_transport REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rosidl_default_generators REQUIRED)

## System dependencies are found with CMake's conventions
find_package(Boost REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(SuiteSparse REQUIRED)
find_package(Threads REQUIRED)

##################
## ROS messages ##
##################
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/FeatureMeasurement.msg"
  "msg/CameraMeasurement.msg"
  "msg/TrackingInfo.msg"
  DEPENDENCIES std_msgs
)

###########
## Build ##
###########

include_directories(
  include
  ${Boost_INCLUDE_DIR}
  ${OpenCV_INCLUDE_DIRS}
  ${SUITESPARSE_INCLUDE_DIRS}
)

# Interfaces generated by this package, needed by the libraries below.
rosidl_get_typesupport_target(cpp_typesupport_target
  ${PROJECT_NAME} "rosidl_typesupport_cpp")

set(msckf_vio_dependencies
  rclcpp
  std_msgs
  sensor_msgs
  nav_msgs
  geometry_msgs
  std_srvs
  tf2
  tf2_ros
  tf2_eigen
  tf2_geometry_msgs
  image_transport
  cv_bridge
  message_filters
  pcl_conversions
)

# Msckf Vio
add_library(msckf_vio_lib
  src/msckf_vio.cpp
  src/utils.cpp
)
ament_target_dependencies(msckf_vio_lib ${msckf_vio_dependencies})
target_include_directories(msckf_vio_lib PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
target_link_libraries(msckf_vio_lib
  ${cpp_typesupport_target}
  Eigen3::Eigen
  ${SUITESPARSE_LIBRARIES}
)

# Image processor
add_library(image_processor
  src/image_processor.cpp
  src/utils.cpp
)
ament_target_dependencies(image_processor ${msckf_vio_dependencies})
target_include_directories(image_processor PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
target_link_libraries(image_processor
  ${cpp_typesupport_target}
  Eigen3::Eigen
  ${OpenCV_LIBS}
)

##############
## Nodes    ##
##############
add_executable(msckf_vio_node src/msckf_vio_node.cpp)
target_link_libraries(msckf_vio_node msckf_vio_lib)

add_executable(image_processor_node src/image_processor_node.cpp)
target_link_libraries(image_processor_node image_processor)

# SGBM and IGEV image processing are CPU-only; the TensorRT runtime is optional.
add_library(stereo_reconstruction_lib
  src/stereo/stereo_matcher.cpp
  src/stereo/igev_processing.cpp
  src/stereo/reconstruction.cpp
)
target_include_directories(stereo_reconstruction_lib PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
  ${OpenCV_INCLUDE_DIRS}
)
target_link_libraries(stereo_reconstruction_lib PUBLIC Eigen3::Eigen ${OpenCV_LIBS})

add_executable(stereo_reconstruction_node src/stereo_reconstruction_node.cpp)
ament_target_dependencies(stereo_reconstruction_node
  rclcpp std_msgs sensor_msgs nav_msgs geometry_msgs std_srvs
  tf2_ros cv_bridge message_filters)
target_link_libraries(stereo_reconstruction_node stereo_reconstruction_lib Threads::Threads)
include(cmake/TensorRTBackend.cmake)

#############
## Install ##
#############

install(TARGETS
  msckf_vio_lib image_processor msckf_vio_node image_processor_node
  stereo_reconstruction_lib stereo_reconstruction_node
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME}
)

install(DIRECTORY include/${PROJECT_NAME}/
  DESTINATION include/${PROJECT_NAME}
)

install(DIRECTORY launch config rviz scripts docs
  DESTINATION share/${PROJECT_NAME}
)

#############
## Testing ##
#############

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)

  # Feature initialization test
  ament_add_gtest(test_feature_init
    test/feature_initialization_test.cpp
  )
  target_include_directories(test_feature_init PRIVATE include)
  target_link_libraries(test_feature_init Eigen3::Eigen)

  # Math utils test
  ament_add_gtest(test_math_utils
    test/math_utils_test.cpp
  )
  target_include_directories(test_math_utils PRIVATE include)
  target_link_libraries(test_math_utils Eigen3::Eigen)

  ament_add_gtest(test_stereo_reconstruction test/stereo_reconstruction_test.cpp)
  target_link_libraries(test_stereo_reconstruction stereo_reconstruction_lib)

  ament_add_gtest(test_igev_processing test/igev_processing_test.cpp)
  target_link_libraries(test_igev_processing stereo_reconstruction_lib)
endif()

ament_package()
